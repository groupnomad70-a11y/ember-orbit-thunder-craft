// ── LIVE PROFIT SIGNAL (Section D) ──
const products = [
  { sold: 'rs',  rev: 'rrev',  cost: 'rcost',  gm: 'rgm',  price: 120, rawcost: 38  },
  { sold: 'cs',  rev: 'crev',  cost: 'ccost',  gm: 'cgm',  price: 320, rawcost: 148 },
  { sold: 'cos', rev: 'corev', cost: 'cocost', gm: 'cogm', price: 80,  rawcost: 22  },
  { sold: 'dcs', rev: 'dcrev', cost: 'dccost', gm: 'dcgm', price: 100, rawcost: 52  },
];

const FLOOR = 6540;

function fmt(n) { return n.toLocaleString('en-KE', {minimumFractionDigits: 0}); }

function recalc() {
  let totRev = 0, totCost = 0, totGM = 0;
  products.forEach(p => {
    const s = parseFloat(document.getElementById(p.sold).value) || 0;
    const rev  = s * p.price;
    const cost = s * p.rawcost;
    const gm   = rev - cost;
    document.getElementById(p.rev).textContent   = fmt(rev);
    document.getElementById(p.cost).textContent  = fmt(cost);
    document.getElementById(p.gm).textContent    = fmt(gm);
    totRev  += rev;
    totCost += cost;
    totGM   += gm;
  });
  document.getElementById('totrev').innerHTML  = '<strong>' + fmt(totRev)  + '</strong>';
  document.getElementById('totcost').innerHTML = '<strong>' + fmt(totCost) + '</strong>';
  document.getElementById('totgm').innerHTML   = '<strong>' + fmt(totGM)   + '</strong>';
  document.getElementById('expected').value    = totRev;

  // Cash variance
  const til = parseFloat(document.getElementById('tillin').value) || 0;
  document.getElementById('cashvar').value = (til - totRev).toFixed(0);

  // Profit signal
  const net = totGM - FLOOR;
  const box = document.getElementById('profit-signal-box');
  const val = document.getElementById('ps-margin');
  const cls = document.getElementById('ps-classify');
  val.textContent = 'KES ' + fmt(Math.abs(net));
  box.style.borderColor = net > 0 ? 'var(--ok)' : net === 0 ? 'var(--warn)' : 'var(--danger)';
  if (net > 0) {
    val.style.color = 'var(--ok)';
    cls.textContent = '✅ Profit — overhead floor cleared by KES ' + fmt(net);
    cls.className = 'ps-signal green';
  } else if (net === 0) {
    val.style.color = 'var(--warn)';
    cls.textContent = '⚠ Exactly break-even — no profit, no loss';
    cls.className = 'ps-signal warn';
  } else {
    val.style.color = 'var(--danger)';
    cls.textContent = '🔴 Loss — KES ' + fmt(Math.abs(net)) + ' short of floor';
    cls.className = 'ps-signal red';
  }
}

// Attach listeners
products.forEach(p => {
  document.getElementById(p.sold).addEventListener('input', recalc);
});
document.getElementById('tillin').addEventListener('input', recalc);
